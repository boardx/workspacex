#!/bin/bash
# Round S5 stack: usage stack.sh createdb|dropdb|seed|redis|model|golden|api|web|pw|northstar|dbproof
# Own postgres cluster 16/wsxs5 on 55475 (AGE + pgvector, UTF8), db s5e2e; ports: kg-eval loopback 29120, golden proxy 29121,
# api 29122, redis 29125, web 29124 (all below the 32768 ephemeral range; the 591xx range was taken by another worker). Same shape as evidence/phase-18/r11a/harness/stack.sh.txt.
WT=/home/user/workspacex/.claude/worktrees/agent-a985e9132f87011f6
S=/tmp/claude-0/-home-user-workspacex/0c400fb8-dfbe-5051-bf5c-eba2226be75a/scratchpad/s5
export PATH=$S/bin:/opt/node22/bin:/usr/local/bin:/usr/bin:/bin
export PGHOST=127.0.0.1 PGPORT=55475 PGPASSWORD=postgres_dev PGDATABASE=s5e2e WORKSPACEX_DB=s5e2e
export WORKSPACEX_MODEL_PROVIDER_PORT=29120 GOLDEN_PORT=29121 WORKSPACEX_API_PORT=29122 WORKSPACEX_WEB_PORT=29124 REDIS_PORT=29125
export KG_EVAL_ORG_ID=org-kg-experience-eval KG_EVAL_PROJECT_ID=project-kg-experience-eval KG_EVAL_OWNER_ID=user-kg-eval-owner
export KG_EVAL_FIXTURE=1 KG_EVAL_AGENT_ID=agent-kg-eval-default KG_EVAL_MODEL_PROVIDER=kg-eval-loopback KG_EVAL_MODEL_ID=kg-eval-grounded
# KG_EVAL_ACCOUNTS: the four accounts of apps/web/e2e/kg-experience-eval/fixture.ts (not copied here).
export KG_EVAL_ACCOUNTS="${KG_EVAL_ACCOUNTS:?JSON array of the four accounts in apps/web/e2e/kg-experience-eval/fixture.ts}"
export PW_CHROME=/opt/pw-browsers/chromium-1194/chrome-linux/chrome
CASES=$WT/evidence/golden/cases.json
case "$1" in
  createdb) exec psql -U postgres -d postgres -c "CREATE DATABASE s5e2e" ;;
  dropdb) exec psql -U postgres -d postgres -c "DROP DATABASE IF EXISTS s5e2e WITH (FORCE)" ;;
  seed) cd $WT/apps/api && npx tsx scripts/seed-kg-experience-eval.ts && cp $WT/evidence/phase-18/r11a/harness/r11a-post-seed.ts.txt scripts/s5-post-seed.ts && npx tsx scripts/s5-post-seed.ts; rm -f $WT/apps/api/scripts/s5-post-seed.ts ;;
  redis) exec redis-server --port $REDIS_PORT --save "" --appendonly no --dir $S/run ;;
  model) cd $WT/apps/api && LOOPBACK_KG_EVAL_PROVIDER_PORT=$WORKSPACEX_MODEL_PROVIDER_PORT LOOPBACK_KG_EVAL_CASES=$CASES exec npx tsx scripts/loopback-kg-eval-model-provider.ts ;;
  golden) cd $WT/apps/api && LOOPBACK_GOLDEN_PROVIDER_PORT=$GOLDEN_PORT LOOPBACK_GOLDEN_UPSTREAM=http://127.0.0.1:$WORKSPACEX_MODEL_PROVIDER_PORT LOOPBACK_GOLDEN_CASES=$CASES exec npx tsx scripts/loopback-golden-model-provider.ts ;;
  api) cd $WT/apps/api && PORT=$WORKSPACEX_API_PORT KERNEL_MODEL_PROVIDER=$KG_EVAL_MODEL_PROVIDER \
      KERNEL_MODEL_BASE_URL=http://127.0.0.1:$GOLDEN_PORT KERNEL_MODEL_API_KEY=kg-eval-loopback-key-not-a-secret \
      KERNEL_KG_EXTRACTION_MODEL_ID=$KG_EVAL_MODEL_ID KERNEL_FOLLOWUP_SUGGESTIONS_MODEL_ID=$KG_EVAL_MODEL_ID \
      KERNEL_MODEL_STREAM_ENABLED=1 KG_EXTRACTION_ENABLED=1 MODEL_CREDENTIAL_KEY=kg-eval-credential-key-not-a-secret \
      exec npx tsx src/main.ts ;;
  web) cd $WT/apps/web && NEXT_PUBLIC_API_URL=http://127.0.0.1:$WORKSPACEX_WEB_PORT NEXT_PUBLIC_API_PATH_PREFIX=/__fullstack_api \
      NEXT_PUBLIC_API_WS_URL=http://127.0.0.1:$WORKSPACEX_API_PORT FULLSTACK_E2E_API_ORIGIN=http://127.0.0.1:$WORKSPACEX_API_PORT \
      NODE_OPTIONS=--max-old-space-size=3072 NEXT_DIST_DIR=.next-s5 NEXT_FONT_GOOGLE_MOCKED_RESPONSES=$WT/apps/web/e2e/support/google-fonts-mock.cjs \
      exec npx next dev ${WEB_TURBO:+--turbo} -p $WORKSPACEX_WEB_PORT ;;
  pw) cd $WT/apps/web && PW_EXECUTABLE=$PW_CHROME exec npx playwright test -c playwright.s5.config.ts "${@:2}" ;;
  northstar) cd $WT/apps/api && exec npx tsx scripts/north-star-restate-rate.ts --org $KG_EVAL_ORG_ID --out $WT/evidence/phase-18/s5/north-star.json ;;
  dbproof) exec psql -U postgres -d s5e2e -f $WT/evidence/phase-18/s5/harness/db-proof.sql ;;
esac
