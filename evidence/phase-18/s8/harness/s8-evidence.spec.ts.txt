/**
 * S8（#4365）真实浏览器证据：平台管理员在「运营状态」看记忆抽取 SLO（卡住的租约告警）→ 打开记忆整合 → 现在整合一次；
 * 本人在大脑页看到「整理记录」、在对话里看到冲突卡 → 撤销这次整理 → 长期记忆恢复原样。
 * 跑在已起好的本地栈上（stack.sh），截图与接口快照写进 evidence/phase-18/s8/。
 */
import { writeFileSync } from "node:fs";
import { expect, test, type Page } from "@playwright/test";
import { getDevModeAccount } from "@repo/dev-mode-accounts";

const OUT = "/home/user/workspacex/.claude/worktrees/agent-a1c38f775f37370ff/evidence/phase-18/s8";
const API = "/__fullstack_api";
const record: Record<string, unknown> = {};

async function login(page: Page, role: "admin" | "consultant" | "lead") {
  const a = getDevModeAccount(role);
  await page.goto("/login");
  await page.getByTestId("login-email").fill(a.email);
  await page.getByTestId("login-password").fill(a.password);
  await page.getByTestId("login-submit").click();
  await page.waitForURL((url) => !url.pathname.startsWith("/login"), { timeout: 300_000, waitUntil: "commit" });
  await page.waitForFunction(() => localStorage.getItem("wsx.sessionToken") !== null, undefined, { timeout: 300_000 });
}
async function api(page: Page, path: string) {
  const token = await page.evaluate(() => localStorage.getItem("wsx.sessionToken"));
  const res = await page.request.get(`${API}${path}`, { headers: { Authorization: `Bearer ${token}` } });
  return { status: res.status(), body: await res.json().catch(() => null) };
}
const personalTexts = async (page: Page) =>
  ((await api(page, "/knowledge-graph/personal")).body as { claims: { statement: string; status: string }[]; objects: { name: string; aliases: string[] }[] });

test("S8: SLO 面板 + 整合 + 撤销", async ({ browser }) => {
  const adminCtx = await browser.newContext();
  const admin = await adminCtx.newPage();
  await login(admin, "admin");
  await admin.goto("/platform-admin/ops-status");
  const slo = admin.getByTestId("admin-extraction-slo");
  await expect(admin.getByTestId("admin-extraction-slo-alert")).toBeVisible({ timeout: 120_000 });
  await expect(admin.getByTestId("admin-extraction-slo-alert-stuck_leases")).toContainText("卡住的租约");
  await expect(admin.getByTestId("admin-consolidation-state")).toHaveText("已关闭");
  record.sloApi = await api(admin, "/platform/knowledge-graph/extraction-slo");
  record.consolidationSettingBefore = await api(admin, "/platform/knowledge-graph/consolidation-setting");
  await admin.getByTestId("admin-platform-extraction").scrollIntoViewIfNeeded();
  await admin.screenshot({ path: `${OUT}/01-admin-slo-alert-and-consolidation-off.png`, fullPage: true });
  await slo.screenshot({ path: `${OUT}/02-admin-slo-panel.png` });

  await admin.getByTestId("admin-consolidation-toggle").click();
  await expect(admin.getByTestId("admin-consolidation-state")).toHaveText("已开启");
  await admin.getByTestId("admin-consolidation-run").click();
  await expect(admin.getByTestId("admin-consolidation-message")).toContainText("整合完成", { timeout: 120_000 });
  record.runMessage = await admin.getByTestId("admin-consolidation-message").innerText();
  await admin.getByTestId("admin-consolidation").screenshot({ path: `${OUT}/03-admin-consolidation-on-and-run.png` });

  const meCtx = await browser.newContext();
  const me = await meCtx.newPage();
  await login(me, "consultant");
  record.personalAfterConsolidation = await personalTexts(me);
  await me.goto("/brain");
  await expect(me.getByTestId("brain-consolidation")).toBeVisible({ timeout: 120_000 });
  await expect(me.getByTestId("brain-consolidation-change-claim_merge")).toContainText("并入");
  await expect(me.getByTestId("brain-consolidation-change-entity_merge")).toContainText("并入");
  await expect(me.getByTestId("brain-consolidation-change-conflict_opened")).toContainText("不一致");
  record.runsBeforeUndo = await api(me, "/knowledge-graph/me/consolidations");
  await me.screenshot({ path: `${OUT}/04-brain-consolidation-record.png`, fullPage: true });

  // 冲突卡：挂在说 60 万的那一轮下面（F16 的卡，不裁决）
  await me.goto("/chat/s8-thr-budget-60");
  const card = me.getByTestId("kg-conflict-card");
  if (await card.isVisible({ timeout: 90_000 }).catch(() => false)) {
    await card.scrollIntoViewIfNeeded();
    await me.screenshot({ path: `${OUT}/05-chat-conflict-card.png`, fullPage: false });
    record.conflictCard = (await card.innerText()).trim();
  } else {
    record.conflictCard = "not visible in chat view (see runsBeforeUndo.conflict_opened)";
  }

  await me.goto("/brain");
  await me.getByTestId("brain-consolidation-undo").click();
  await expect(me.getByTestId("brain-consolidation-run-state")).toHaveText("已撤销", { timeout: 60_000 });
  record.runsAfterUndo = await api(me, "/knowledge-graph/me/consolidations");
  record.personalAfterUndo = await personalTexts(me);
  await me.screenshot({ path: `${OUT}/06-brain-after-undo.png`, fullPage: true });

  // 其他人（lead）看不到这个人的整理记录
  const otherCtx = await browser.newContext();
  const other = await otherCtx.newPage();
  await login(other, "lead");
  record.otherUserRuns = await api(other, "/knowledge-graph/me/consolidations");
  record.otherUserSlo = await api(other, "/platform/knowledge-graph/extraction-slo");

  writeFileSync(`${OUT}/journey.json`, `${JSON.stringify(record, null, 2)}\n`);
});
