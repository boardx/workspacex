/**
 * Issue #4278 round 5: a decision promoted to personal space is force-recalled in a NEW personal thread's
 * unrelated turn, and (sign-off question (c)) is NOT carried into the user's own NEW project thread.
 *
 * Model is the deterministic KG-eval loopback (`apps/api/scripts/loopback-kg-eval-model-provider.ts`), fed a
 * one-sentence corpus so that extracting exactly 「我决定关注 211 高校」 returns one decision claim (it plays "an
 * extraction model that read this sentence right"); chat answers only echo the 【记忆】 block the executor actually
 * handed the model that turn. Everything else — extraction worker, promotion, PgKnowledgeRecall, fuseRecall,
 * `kg_turn_recalls`, the answer footer — is the real product path.
 */
import { mkdirSync, writeFileSync } from "node:fs";
import path from "node:path";
import { expect, test, type Page } from "@playwright/test";
import { KG_EVAL } from "../kg-experience-eval/fixture";
import { apiGet, login, newThread, openMemoryPanel, say, turnBlock } from "../kg-experience-eval/eval-helpers";

const OUT = path.resolve(__dirname, "../../../../evidence/phase-18/r05");
mkdirSync(OUT, { recursive: true });
const DECISION = "我决定关注 211 高校";
const UNRELATED = "开始写报告吧";
const record: Record<string, unknown> = {};
const shot = (page: Page, name: string) => page.screenshot({ path: path.join(OUT, `${name}.png`), fullPage: false });
const dump = () => writeFileSync(path.join(OUT, "journey.json"), `${JSON.stringify(record, null, 2)}\n`);

interface Claim { id: string; statement: string; status: string; kind?: string }
interface Recalled { claimId: string; statement: string; scope: string; channels: string[]; score: number }
interface TurnMemory { recalled: Recalled[] }

async function threadClaims(page: Page, threadId: string) {
  return (await apiGet(page, `/knowledge-graph/threads/${threadId}`)).body as { extractionActive: boolean; claims: Claim[] };
}
async function turnMemory(page: Page, threadId: string, answerId: string) {
  const r = await apiGet(page, `/knowledge-graph/threads/${threadId}/messages/${answerId}/memory`);
  expect(r.status).toBe(200);
  return r.body as TurnMemory;
}
async function personal(page: Page) {
  return ((await apiGet(page, "/knowledge-graph/personal")).body as { claims: Claim[] }).claims
    .map((c) => ({ id: c.id, statement: c.statement, status: c.status }));
}
async function waitClaim(page: Page, threadId: string, statement: string) {
  let hit: Claim | undefined;
  await expect.poll(async () => {
    hit = (await threadClaims(page, threadId)).claims.find((c) => c.statement === statement);
    return hit !== undefined;
  }, { timeout: 60_000 }).toBe(true);
  return hit!;
}
/** 「新对话」 in the project screen (`/chat?projectId=…`): same button, project-scoped list. */
async function newProjectThread(page: Page, projectId: string): Promise<string> {
  await page.goto(`/chat?projectId=${encodeURIComponent(projectId)}`);
  const create = page.getByTestId("chat-thread-create");
  await expect(create).toBeVisible();
  const before = new URL(page.url()).pathname;
  await create.click();
  await page.waitForURL((url) => /^\/chat\/[^/]+$/.test(url.pathname) && url.pathname !== before);
  const threadId = decodeURIComponent(new URL(page.url()).pathname.split("/").at(-1)!);
  await expect(page.getByTestId("copilotkit-v2-input")).toBeVisible();
  await expect(page.getByTestId("copilot-assistant-message")).toHaveCount(0, { timeout: 15_000 });
  return threadId;
}

test.describe.configure({ mode: "serial" });
let page: Page;
const ids: Record<string, string> = {};

test.beforeAll(async ({ browser }) => {
  page = await browser.newPage();
  record.ids = ids;
  await login(page, KG_EVAL.owner);
});
test.afterAll(() => dump());

test("step 1: in a personal thread, the decision is extracted", async () => {
  ids.threadP1 = await newThread(page);
  const t = await say(page, DECISION);
  record.step1_turn = { answerId: t.answerId, text: t.text };
  const chip = page.getByTestId("kg-extraction-feedback");
  await expect(chip).toBeVisible({ timeout: 60_000 });
  await expect(chip).toContainText(`已记下：${DECISION}`);
  record.step1_chip = (await chip.innerText()).trim();
  const claim = await waitClaim(page, ids.threadP1, DECISION);
  expect(claim.kind).toBe("decision");
  ids.sourceClaim = claim.id;
  await shot(page, "01-extracted-chip");
});

test("step 2: promote it to personal space from the memory panel", async () => {
  await openMemoryPanel(page);
  await page.getByTestId("kg-promote-enter").click();
  await page.getByTestId(`kg-claim-select-${ids.sourceClaim}`).check();
  await page.getByTestId("kg-promote-submit").click();
  await expect(page.getByTestId("kg-promotion-summary")).toContainText("已记到长期记忆 1 条");
  await shot(page, "02-promoted");
  const p = await personal(page);
  record.step2_personal = p;
  const promoted = p.find((c) => c.statement === DECISION);
  expect(promoted).toBeTruthy();
  ids.personalClaim = promoted!.id;
});

test("step 3: a NEW personal thread, unrelated turn 「开始写报告吧」, recalls the decision labelled personal space", async () => {
  ids.threadP2 = await newThread(page);
  expect(ids.threadP2).not.toBe(ids.threadP1);
  const t = await say(page, UNRELATED);
  const mem = await turnMemory(page, ids.threadP2, t.answerId);
  const footer = turnBlock(t.answer);
  await expect(footer).toContainText(DECISION, { timeout: 30_000 });
  record.step3 = { answerId: t.answerId, text: t.text, recalled: mem.recalled, footer: (await footer.innerText()).trim() };
  const hit = mem.recalled.find((r) => r.claimId === ids.personalClaim);
  expect(hit).toMatchObject({ statement: DECISION, scope: "personal", channels: ["claim"], score: 0 });
  expect(t.text).toContain(DECISION);
  expect(t.text).toContain("来自个人空间知识");
  await t.answer.scrollIntoViewIfNeeded();
  await shot(page, "03-new-personal-thread-recalls");
});

// Observation only, not a done criterion: current behaviour, pending usecases.md #4278 sign-off question (c).
test("step 4 (current behaviour, pending sign-off (c)): a NEW project thread, same unrelated turn, does not get the personal decision", async () => {
  ids.threadProject = await newProjectThread(page, KG_EVAL.projectId);
  const t = await say(page, UNRELATED);
  const mem = await turnMemory(page, ids.threadProject, t.answerId);
  record.step4 = { answerId: t.answerId, text: t.text, recalled: mem.recalled, footer: (await turnBlock(t.answer).innerText()).trim() };
  expect(mem.recalled.map((r) => r.claimId)).not.toContain(ids.personalClaim);
  expect(mem.recalled.map((r) => r.statement)).not.toContain(DECISION);
  expect(t.text).not.toContain("211");
  await expect(turnBlock(t.answer)).not.toContainText(DECISION);
  await shot(page, "04-current-behaviour-project-thread-no-recall");
  // Non-vacuity: the personal decision is still live right now.
  expect((await personal(page)).map((c) => c.id)).toContain(ids.personalClaim);
});
